✔ 9.1 Program.cs — Full Pipeline Example
csharp
using Opentail.Stingray;
using Opentail.Stingray.Diffusion;
using Opentail.Stingray.Diffusion.Loading;
using Opentail.Stingray.Diffusion.ControlNet;

public static class Program
{
    public static void Main()
    {
        // 1) Choose backend (ggml, CUDA, DirectML, Vulkan, Metal)
        IBackend backend = BackendFactory.CreateDefault();

        // 2) Load SD 1.5 weights (safetensors or GGUF)
        var registry = new ModelRegistry(backend);
        registry.LoadSd15("sd15.safetensors");

        // 3) Choose scheduler
        IScheduler scheduler = new EulerAncestralScheduler(backend, numSteps: 30);

        // 4) Build pipeline
        var pipeline = new DiffusionPipeline(registry, scheduler);

        // 5) Tokenize prompt
        ITensor tokens = backend.Tokenize(
            "a beautiful watercolor painting of a fox in a forest");

        // 6) Optional ControlNet hint (e.g., canny edges)
        ITensor? controlHint = null;
        // controlHint = backend.LoadImageAsTensor("canny.png");

        // 7) Run inference
        ITensor image = pipeline.Generate(
            tokens,
            width: 512,
            height: 512,
            steps: 30,
            controlHint: controlHint);

        // 8) Save output
        backend.SaveImage(image, "output.png");
    }
}
This is a real, complete pipeline.
If your backend implements the required ops, this will generate an SD 1.5 image.
