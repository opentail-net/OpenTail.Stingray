VAE ResBlock

public sealed class VaeResBlock
{
    private readonly GroupNorm _norm1;
    private readonly Conv2d _conv1;
    private readonly GroupNorm _norm2;
    private readonly Conv2d _conv2;

    public VaeResBlock(int channels)
    {
        _norm1 = new GroupNorm(32, channels);
        _conv1 = new Conv2d(channels, channels, (3, 3), (1, 1), (1, 1));

        _norm2 = new GroupNorm(32, channels);
        _conv2 = new Conv2d(channels, channels, (3, 3), (1, 1), (1, 1));
    }

    public ITensor Forward(ITensor x)
    {
        var h = _norm1.Forward(x);
        h = Ops.Silu(h);
        h = _conv1.Forward(h);

        h = _norm2.Forward(h);
        h = Ops.Silu(h);
        h = _conv2.Forward(h);

        return Ops.Add(h, x);
    }
}

public sealed class VaeUpsample
{
    private readonly Conv2d _conv;

    public VaeUpsample(int channels)
    {
        _conv = new Conv2d(channels, channels, (3, 3), (1, 1), (1, 1));
    }

    public ITensor Forward(ITensor x)
    {
        var up = Backend.UpsampleNearest2x(x);

public sealed class VaeDecoder : IVaeDecoder
{
    private readonly IBackend _backend;

    private readonly Conv2d _inConv;

    private readonly VaeResBlock _block1;
    private readonly VaeResBlock _block2;
    private readonly VaeUpsample _up1;

    private readonly VaeResBlock _block3;
    private readonly VaeResBlock _block4;
    private readonly VaeUpsample _up2;

    private readonly VaeResBlock _block5;
    private readonly VaeResBlock _block6;
    private readonly VaeUpsample _up3;

    private readonly GroupNorm _finalNorm;
    private readonly Conv2d _finalConv;

    public VaeDecoder(IBackend backend)
    {
        _backend = backend;

        // SD 1.5 VAE uses 4 → 256 channels
        _inConv = new Conv2d(4, 256, (3, 3), (1, 1), (1, 1));

        _block1 = new VaeResBlock(256);
        _block2 = new VaeResBlock(256);
        _up1 = new VaeUpsample(256);

        _block3 = new VaeResBlock(256);
        _block4 = new VaeResBlock(256);
        _up2 = new VaeUpsample(256);

        _block5 = new VaeResBlock(256);
        _block6 = new VaeResBlock(256);
        _up3 = new VaeUpsample(256);

        _finalNorm = new GroupNorm(32, 256);
        _finalConv = new Conv2d(256, 3, (3, 3), (1, 1), (1, 1));
    }

    public ITensor Decode(ITensor latents)
    {
        // SD latents are scaled by 1/0.18215
        var x = Backend.MulScalar(latents, 1f / 0.18215f);

        // Input conv
        x = _inConv.Forward(x);

        // Stage 1
        x = _block1.Forward(x);
        x = _block2.Forward(x);
        x = _up1.Forward(x);

        // Stage 2
        x = _block3.Forward(x);
        x = _block4.Forward(x);
        x = _up2.Forward(x);

        // Stage 3
        x = _block5.Forward(x);
        x = _block6.Forward(x);
        x = _up3.Forward(x);

        // Final
        x = _finalNorm.Forward(x);
        x = Ops.Silu(x);
        x = _finalConv.Forward(x);

        // SD VAE outputs tanh → [-1, 1]
        return Backend.Tanh(x);
    }
}
