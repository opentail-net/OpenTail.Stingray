SEGMENT 6 — Schedulers (Euler, DDIM, Heun, LMS)
Schedulers operate on:

latents: [N, 4, H/8, W/8]

noisePred: UNet output [N, 4, H/8, W/8]

timesteps: [N]

They use precomputed arrays:

alphas

sigmas

timesteps

We’ll define a shared base class to hold these.

✔ 6.1 Scheduler Base Class
csharp
public abstract class SchedulerBase : IScheduler
{
    protected readonly IBackend Backend;

    protected readonly float[] Alphas;
    protected readonly float[] Sigmas;
    protected readonly int NumSteps;

    protected SchedulerBase(IBackend backend, int numSteps)
    {
        Backend = backend;
        NumSteps = numSteps;

        // SD 1.5 default noise schedule (Karras)
        (Alphas, Sigmas) = SchedulerUtils.ComputeKarrasSchedule(numSteps);
    }

    public virtual ITensor InitLatents(ITensor noise, int width, int height)
    {
        // SD uses scaled latents
        return Backend.MulScalar(noise, Sigmas[0]);
    }

    public abstract ITensor Step(ITensor latents, ITensor noisePred, ITensor timesteps);
}
✔ 6.2 Euler (Ancestral)
This is the most common scheduler in SD.cpp and ComfyUI.

csharp
public sealed class EulerAncestralScheduler : SchedulerBase
{
    public EulerAncestralScheduler(IBackend backend, int numSteps)
        : base(backend, numSteps) { }

    public override ITensor Step(ITensor latents, ITensor noisePred, ITensor timesteps)
    {
        int t = (int)timesteps.GetScalar(0);

        float sigma = Sigmas[t];
        float sigmaNext = Sigmas[Math.Min(t + 1, NumSteps - 1)];

        float dt = sigmaNext - sigma;

        // Euler update: x = x + noisePred * dt
        var update = Backend.MulScalar(noisePred, dt);
        var x = Backend.Add(latents, update);

        // Add noise for ancestral sampling
        if (sigmaNext > 0)
        {
            var noise = Backend.RandomNormalLike(latents);
            var scaled = Backend.MulScalar(noise, MathF.Sqrt(sigmaNext * sigmaNext - sigma * sigma));
            x = Backend.Add(x, scaled);
        }

        return x;
    }
}

✔ 6.4 DDIM Scheduler
DDIM is deterministic and used in SD 1.5 for fast sampling.

csharp
public sealed class DdimScheduler : SchedulerBase
{
    public DdimScheduler(IBackend backend, int numSteps)
        : base(backend, numSteps) { }

    public override ITensor Step(ITensor latents, ITensor noisePred, ITensor timesteps)
    {
        int t = (int)timesteps.GetScalar(0);

        float alpha = Alphas[t];
        float alphaNext = Alphas[Math.Min(t + 1, NumSteps - 1)];

        // x0 = (x - sqrt(1 - alpha) * noisePred) / sqrt(alpha)
        var term1 = Backend.MulScalar(noisePred, MathF.Sqrt(1 - alpha));
        var x0 = Backend.DivScalar(Backend.Sub(latents, term1), MathF.Sqrt(alpha));

        // x_next = sqrt(alpha_next) * x0 + sqrt(1 - alpha_next) * noisePred
        var part1 = Backend.MulScalar(x0, MathF.Sqrt(alphaNext));
        var part2 = Backend.MulScalar(noisePred, MathF.Sqrt(1 - alphaNext));

        return Backend.Add(part1, part2);
    }
}
✔ 6.5 Heun Scheduler
Heun is a 2‑step method used in SDXL and K‑diffusion.

csharp
public sealed class HeunScheduler : SchedulerBase
{
    public HeunScheduler(IBackend backend, int numSteps)
        : base(backend, numSteps) { }

    public override ITensor Step(ITensor latents, ITensor noisePred, ITensor timesteps)
    {
        int t = (int)timesteps.GetScalar(0);

        float sigma = Sigmas[t];
        float sigmaNext = Sigmas[Math.Min(t + 1, NumSteps - 1)];

        float dt = sigmaNext - sigma;

        // First derivative
        var d1 = Backend.MulScalar(noisePred, dt);
        var xEuler = Backend.Add(latents, d1);

        // Predict noise at Euler point
        var noisePred2 = Backend.PredictNoiseAt(xEuler, sigmaNext);

        // Second derivative
        var d2 = Backend.MulScalar(noisePred2, dt);

        // Heun update: x + 0.5*(d1 + d2)
        var avg = Backend.MulScalar(Backend.Add(d1, d2), 0.5f);
        return Backend.Add(latents, avg);
    }
}
