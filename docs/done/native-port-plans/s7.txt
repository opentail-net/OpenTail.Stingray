🌉 SEGMENT 7 — ControlNet Integration (C#)
ControlNet is essentially:

A copy of the UNet downsampling path,

With zero‑convs that produce control tensors,

Which are added into the UNet skip connections.

We’ll implement the SD‑1.5 version first (the simplest), but structure it so SDXL can be added later.

7.1 ControlNetBlock.cs
This mirrors the SD‑1.5 architecture:

Down blocks

Attention blocks

Zero‑convs

Middle block

Output control tensors

csharp
namespace Opentail.Stingray.Diffusion.ControlNet
{
    public sealed class ControlNetBlock
    {
        private readonly IBackend _backend;

        private readonly Linear _timeEmbed1;
        private readonly Linear _timeEmbed2;

        private readonly List<IUNetBlock> _downBlocks = new();
        private readonly List<IUNetBlock> _downSamples = new();
        private readonly List<Conv2d> _zeroConvs = new();

        private readonly IUNetBlock _midRes1;
        private readonly IUNetBlock _midAttn;
        private readonly IUNetBlock _midRes2;
        private readonly Conv2d _midZeroConv;

        public ControlNetBlock(IBackend backend)
        {
            _backend = backend;

            // Time embedding
            _timeEmbed1 = new Linear(320, 1280);
            _timeEmbed2 = new Linear(1280, 1280);

            int[] chMult = { 1, 2, 4, 4 };
            int baseChannels = 320;
            int ch = baseChannels;

            // Down path
            foreach (var mult in chMult)
            {
                int outCh = baseChannels * mult;

                _downBlocks.Add(new ResBlock(ch, outCh, 1280));
                _zeroConvs.Add(new Conv2d(outCh, outCh, (1, 1), (1, 1), (0, 0)));

                if (mult == 1 || mult == 2 || mult == 4)
                    _downBlocks.Add(new AttentionBlock(outCh));

                ch = outCh;

                if (mult != chMult.Last())
                {
                    _downSamples.Add(new Downsample(ch));
                    _zeroConvs.Add(new Conv2d(ch, ch, (1, 1), (1, 1), (0, 0)));
                }
            }

            // Middle
            _midRes1 = new ResBlock(ch, ch, 1280);
            _midAttn = new AttentionBlock(ch);
            _midRes2 = new ResBlock(ch, ch, 1280);
            _midZeroConv = new Conv2d(ch, ch, (1, 1), (1, 1), (0, 0));
        }

        public IReadOnlyList<ITensor> Forward(
            ITensor x,
            ITensor hint,
            ITensor timesteps,
            ITensor context)
        {
            // 1) Time embedding
            var tEmb = Ops.TimestepEmbedding(_backend, timesteps, 320);
            tEmb = Ops.Silu(_timeEmbed1.Forward(tEmb));
            tEmb = _timeEmbed2.Forward(tEmb);

            // 2) Add hint to input
            var h = Ops.Add(x, hint);

            var outputs = new List<ITensor>();

            int zeroIndex = 0;
            int downIndex = 0;

            // 3) Down path
            foreach (var block in _downBlocks)
            {
                h = block.Forward(h, tEmb, context);

                var z = _zeroConvs[zeroIndex++].Forward(h);
                outputs.Add(z);

                if (downIndex < _downSamples.Count)
                {
                    h = _downSamples[downIndex].Forward(h, tEmb, context);

                    var z2 = _zeroConvs[zeroIndex++].Forward(h);
                    outputs.Add(z2);

                    downIndex++;
                }
            }

            // 4) Middle block
            h = _midRes1.Forward(h, tEmb, context);
            h = _midAttn.Forward(h, tEmb, context);
            h = _midRes2.Forward(h, tEmb, context);

            outputs.Add(_midZeroConv.Forward(h));

            return outputs;
        }
    }
}
7.2 ControlNetRunner.cs
This runs the block and caches the guided hint.

csharp
namespace Opentail.Stingray.Diffusion.ControlNet
{
    public sealed class ControlNetRunner
    {
        private readonly ControlNetBlock _block;
        private readonly IBackend _backend;

        private ITensor? _cachedHint;

        public ControlNetRunner(IBackend backend)
        {
            _backend = backend;
            _block = new ControlNetBlock(backend);
        }

        public IReadOnlyList<ITensor> Compute(
            ITensor x,
            ITensor hint,
            ITensor timesteps,
            ITensor context)
        {
            // Cache guided hint if needed
            if (_cachedHint == null)
                _cachedHint = hint;

            return _block.Forward(x, _cachedHint, timesteps, context);
        }
    }
}

public ITensor Generate(
    ITensor promptTokens,
    int width,
    int height,
    int numSteps,
    ITensor? controlHint = null)
{
    var context = _textEncoder.Encode(promptTokens);

    var noise = _backend.RandomNormalLikeShape(
        batch: promptTokens.Shape[0],
        channels: 4,
        height: height / 8,
        width: width / 8);

    var latents = _scheduler.InitLatents(noise, width, height);

    IReadOnlyList<ITensor>? control = null;

    if (controlHint != null)
    {
        var controlNet = new ControlNetRunner(_backend);
        control = controlNet.Compute(latents, controlHint, _backend.MakeTimesteps(0, promptTokens.Shape[0]), context);
    }

    for (int i = 0; i < numSteps)
    {
        var t = _backend.MakeTimesteps(i, promptTokens.Shape[0]);
        var noisePred = _unet.PredictNoise(latents, t, context, control);
        latents = _scheduler.Step(latents, noisePred, t);
    }

    return _vae.Decode(latents);
}
