⭐ 8.1 WeightLoader.cs
This class loads tensors from any supported format and returns them as Stingray tensors.

It does not know anything about UNet or CLIP — it only loads raw tensors.

csharp
namespace Opentail.Stingray.Diffusion.Loading
{
    public sealed class WeightLoader
    {
        private readonly IBackend _backend;

        public WeightLoader(IBackend backend)
        {
            _backend = backend;
        }

        public Dictionary<string, ITensor> LoadFromSafetensors(string path)
        {
            var raw = SafetensorsReader.Read(path); // returns float[] + shape
            var dict = new Dictionary<string, ITensor>();

            foreach (var kv in raw)
            {
                var name = kv.Key;
                var (data, shape) = kv.Value;

                var tensor = _backend.CreateTensor(shape);
                tensor.Assign(data);

                dict[name] = tensor;
            }

            return dict;
        }

        public Dictionary<string, ITensor> LoadFromGguf(string path)
        {
            var raw = GgufReader.Read(path); // returns float[] + shape
            var dict = new Dictionary<string, ITensor>();

            foreach (var kv in raw)
            {
                var name = kv.Key;
                var (data, shape) = kv.Value;

                var tensor = _backend.CreateTensor(shape);
                tensor.Assign(data);

                dict[name] = tensor;
            }

            return dict;
        }
    }
}

⭐ 8.2 TensorAssignment.cs
This class maps weight names → model block parameters.

It’s the “router” that connects loaded weights to your UNet, CLIP, VAE, ControlNet.

csharp
namespace Opentail.Stingray.Diffusion.Loading
{
    public sealed class TensorAssignment
    {
        public void AssignLinear(Linear linear, ITensor weight, ITensor? bias)
        {
            linear.Weight.Assign(weight);
            if (bias != null)
                linear.Bias.Assign(bias);
        }

        public void AssignConv2d(Conv2d conv, ITensor weight, ITensor? bias)
        {
            conv.Weight.Assign(weight);
            if (bias != null)
                conv.Bias.Assign(bias);
        }

        public void AssignGroupNorm(GroupNorm gn, ITensor gamma, ITensor beta)
        {
            gn.Gamma.Assign(gamma);
            gn.Beta.Assign(beta);
        }

        public void AssignLayerNorm(LayerNorm ln, ITensor gamma, ITensor beta)
        {
            ln.Gamma.Assign(gamma);
            ln.Beta.Assign(beta);
        }
    }
}
This is intentionally simple — it’s the glue layer.

⭐ 8.4 Pipeline Integration
Your pipeline now becomes:

csharp
public sealed class DiffusionPipeline
{
    private readonly ModelRegistry _models;
    private readonly IScheduler _scheduler;

    public DiffusionPipeline(ModelRegistry models, IScheduler scheduler)
    {
        _models = models;
        _scheduler = scheduler;
    }

    public ITensor Generate(ITensor tokens, int width, int height, int steps, ITensor? controlHint = null)
    {
        var clip = _models.TextEncoder;
        var unet = _models.UNet;
        var vae = _models.Vae;
        var control = _models.ControlNet;

        var context = clip.Encode(tokens);

        var noise = _models.UNet.Backend.RandomNormalLikeShape(
            batch: tokens.Shape[0],
            channels: 4,
            height: height / 8,
            width: width / 8);

        var latents = _scheduler.InitLatents(noise, width, height);

        IReadOnlyList<ITensor>? controlTensors = null;

        if (controlHint != null && control != null)
        {
            controlTensors = control.Compute(latents, controlHint, _models.UNet.Backend.MakeTimesteps(0, tokens.Shape[0]), context);
        }

        for (int i = 0; i < steps)
        {
            var t = _models.UNet.Backend.MakeTimesteps(i, tokens.Shape[0]);
            var noisePred = unet.PredictNoise(latents, t, context, controlTensors);
            latents = _scheduler.Step(latents, noisePred, t);
        }

        return vae.Decode(latents);
    }
}
